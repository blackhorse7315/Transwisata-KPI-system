rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // OPERATIONS MANUAL PORTAL
    function portalMember() {
      return request.auth != null && request.auth.token.firebase.sign_in_provider != 'anonymous'
        && exists(/databases/$(database)/documents/operationsManualPortal/transwisata/members/$(request.auth.uid))
        && get(/databases/$(database)/documents/operationsManualPortal/transwisata/members/$(request.auth.uid)).data.active == true
        && get(/databases/$(database)/documents/operationsManualPortal/transwisata/members/$(request.auth.uid)).data.role in ['admin', 'user'];
    }
    match /operationsManualPortal/{workspace}/workspace/current {
      allow read: if workspace == 'transwisata' && portalMember();
    }
    match /operationsManualPortal/{workspace}/versions/{version}/chunks/{chunk} {
      allow read: if workspace == 'transwisata' && portalMember();
    }
    match /operationsManualPortal/{workspace}/{document=**} {
      allow write: if false;
    }
    // TRANSWISATA e-KPI V3.4
    function ekSigned() { return request.auth != null && request.auth.token.firebase.sign_in_provider != 'anonymous'; }
    function ekProfile() { return get(/databases/$(database)/documents/ekpi_users/$(request.auth.uid)).data; }
    function ekApproved() { return ekSigned() && ekProfile().status == 'Approved'; }
    function ekAdmin() { return ekApproved() && ekProfile().admin == true; }
    match /ekpi_users/{uid} {
     allow read: if ekSigned() && (request.auth.uid == uid || ekAdmin());
     allow create: if ekSigned() && request.auth.uid == uid && request.resource.data.keys().hasOnly(['name','job','email','admin','status','scope']) && request.resource.data.admin == false && request.resource.data.status == 'Pending' && request.resource.data.scope == [] && request.resource.data.email == request.auth.token.email;
     allow update: if ekAdmin() && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['scope','status','allEmployees','canEvaluate','canReview','canApprove']);
    }
    match /ekpi_employees/{id} {
     allow read: if ekAdmin() || (ekApproved() && (ekProfile().get('allEmployees',false) == true || resource.data.role in ekProfile().scope
            || resource.data.get('roles', []).hasAny(ekProfile().scope)));
     allow write: if ekAdmin();
    }
    match /ekpi_evaluators/{id} { allow read, write: if ekAdmin(); }
    match /ekpi_settings/{id} { allow read: if ekApproved(); allow write: if ekAdmin(); }
    function validOwnSign(s) { return s is string && s.size() > 0 && s.size() <= 250000 && s.matches('data:image/jpeg;base64,[A-Za-z0-9+/=]+'); }
    function ownSignature(s) { return validOwnSign(s) && s == get(/databases/$(database)/documents/ekpi_signatures/$(request.auth.uid)).data.sign; }
    match /ekpi_signatures/{uid} {
      allow read: if ekApproved() && request.auth.uid == uid;
      allow create, update: if ekApproved() && request.auth.uid == uid && request.resource.data.keys().hasOnly(['sign']) && validOwnSign(request.resource.data.sign);
    }
    match /ekpi_reviews/{id} {
     allow read: if ekAdmin() || (ekApproved() && (resource.data.accountEmail == request.auth.token.email || ekProfile().get('canReview',false) == true || ekProfile().get('canApprove',false) == true));
     allow create: if ekAdmin() || (ekApproved() && request.resource.data.accountEmail == request.auth.token.email && request.resource.data.status == 'Submitted' && request.resource.data.evaluator == ekProfile().name && request.resource.data.evaluatorJob == ekProfile().job && ekProfile().get('canEvaluate',true) == true && (ekProfile().get('allEmployees',false) == true || request.resource.data.role in ekProfile().scope) && (get(/databases/$(database)/documents/ekpi_employees/$(request.resource.data.emp)).data.role == request.resource.data.role || request.resource.data.role in get(/databases/$(database)/documents/ekpi_employees/$(request.resource.data.emp)).data.get('roles', [])) && ownSignature(request.resource.data.evaluatorSign) && request.resource.data.get('approverSign','') == '');
     allow delete: if ekAdmin();
     allow update: if ekAdmin() || (ekApproved() && resource.data.accountEmail != request.auth.token.email
     && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status','decisionNote','decisionUid','decisionName','decisionJob','decisionAt','approverSign'])
     && request.resource.data.decisionUid == request.auth.uid && request.resource.data.decisionName == ekProfile().name && request.resource.data.decisionJob == ekProfile().job
     && ((request.resource.data.status == 'Approved' && ekProfile().job == 'President Director' && ownSignature(request.resource.data.approverSign)) || ((request.resource.data.status != 'Approved' || ekProfile().job != 'President Director') && request.resource.data.get('approverSign','') == ''))
     && request.resource.data.decisionAt == request.time && request.resource.data.decisionNote is string && request.resource.data.decisionNote.size() > 0
     && ((ekProfile().get('canReview',false) == true && resource.data.status == 'Submitted' && request.resource.data.status in ['Reviewed','Returned']) || (ekProfile().get('canApprove',false) == true && resource.data.status in ['Submitted','Reviewed'] && request.resource.data.status in ['Approved','Returned'])));
    }
    
    match /ekpi_reviewers/{id} { allow read, write: if ekAdmin(); }
    // PRESERVE EXISTING APPLICATIONS
    match /{collection}/{document=**} {
      allow read, write: if collection != 'operationsManualPortal'
        && !collection.matches('ekpi_.*')
        && request.auth != null;
    }
  }
}
